cmake_minimum_required(VERSION 3.24)
project(casper_spike LANGUAGES CXX)
enable_testing()
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_library(casper_animation_core STATIC "${REPOSITORY_ROOT}/Extension/AnimationStudio/anim_core.cpp")
target_include_directories(casper_animation_core PRIVATE "${REPOSITORY_ROOT}")
target_compile_features(casper_animation_core PRIVATE cxx_std_20)
# Upstream anim_core contains unused JSON helpers; keep its existing source
# intact and apply the strict Arcade warning policy to the spike/test code.
add_executable(casper_spike_tests flatland_tests.cpp)
target_link_libraries(casper_spike_tests PRIVATE casper_animation_core)
target_include_directories(casper_spike_tests PRIVATE "${REPOSITORY_ROOT}")
target_compile_features(casper_spike_tests PRIVATE cxx_std_20)
include("${CMAKE_CURRENT_SOURCE_DIR}/../manual_test_warnings.cmake")
dingosdk_arcade_manual_test_warnings(casper_spike_tests)
add_test(NAME casper_spike COMMAND casper_spike_tests)
# Compile the actual outer input gate as well as the portable mailbox. A loaded
# authored catalog must not be mistaken for an active competing playback.
set(flatland_source "${REPOSITORY_ROOT}/Extension/ArcadeMode/arcade_mode.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${flatland_source}")
file(READ "${flatland_source}" flatland_mode)
string(FIND "${flatland_mode}" "        auto flatland=flatland_context(" flatland_begin)
string(FIND "${flatland_mode}" "        const auto flatland_binding=" flatland_end)
if(flatland_begin LESS 0 OR flatland_end LESS flatland_begin)
    message(FATAL_ERROR "Casper runtime admission source boundary changed")
endif()
math(EXPR flatland_length "${flatland_end}-${flatland_begin}")
string(SUBSTRING "${flatland_mode}" ${flatland_begin} ${flatland_length} flatland_admission)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/runtime_admission.inc" "${flatland_admission}")
add_executable(casper_runtime_tests runtime_tests.cpp "${REPOSITORY_ROOT}/Extension/ArcadeMode/arcade_custom_flatland_runtime.cpp")
target_link_libraries(casper_runtime_tests PRIVATE casper_animation_core)
target_include_directories(casper_runtime_tests PRIVATE "${REPOSITORY_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(casper_runtime_tests PRIVATE cxx_std_20)
target_compile_definitions(casper_runtime_tests PRIVATE DINGOSDK_DEV_TOOLS)
dingosdk_arcade_manual_test_warnings(casper_runtime_tests)
add_test(NAME casper_runtime COMMAND casper_runtime_tests)
add_executable(casper_release_tests release_tests.cpp "${REPOSITORY_ROOT}/Extension/ArcadeMode/arcade_custom_flatland_runtime.cpp")
target_include_directories(casper_release_tests PRIVATE "${REPOSITORY_ROOT}")
target_compile_features(casper_release_tests PRIVATE cxx_std_20)
dingosdk_arcade_manual_test_warnings(casper_release_tests)
# The full dev build defines this directory-wide. This target deliberately
# compiles the same runtime source as a release, without animation endpoints.
if(MSVC)
    target_compile_options(casper_release_tests PRIVATE /UDINGOSDK_DEV_TOOLS)
else()
    target_compile_options(casper_release_tests PRIVATE -UDINGOSDK_DEV_TOOLS)
endif()
add_test(NAME casper_release COMMAND casper_release_tests)
